    const matScaffold = new THREE.MeshStandardMaterial({ color: 0x4a90d9, metalness: 0.35, roughness: 0.45, transparent: true, opacity: 0.92 });

    const matInner    = new THREE.MeshStandardMaterial({ color: 0x3a7bc8, metalness: 0.4, roughness: 0.4, transparent: true, opacity: 0.88 });

    const matFG       = new THREE.MeshStandardMaterial({ color: 0xe8a838, metalness: 0.1, roughness: 0.7, transparent: true, opacity: 0.55, side: THREE.DoubleSide });

    const matFilament = new THREE.MeshStandardMaterial({ color: 0x6ab0e8, metalness: 0.2, roughness: 0.6, transparent: true, opacity: 0.75 });

    const matBasket   = new THREE.MeshStandardMaterial({ color: 0x5a9ad0, metalness: 0.25, roughness: 0.55, transparent: true, opacity: 0.8 });

    const matMembrane = new THREE.MeshStandardMaterial({ color: 0x2a3a55, metalness: 0.15, roughness: 0.85, transparent: true, opacity: 0.55, side: THREE.DoubleSide });

    const matCargoIn  = new THREE.MeshStandardMaterial({ color: 0x5cd47d, emissive: 0x1a5a2a, emissiveIntensity: 0.4, metalness: 0.1, roughness: 0.3 });

    const matCargoOut = new THREE.MeshStandardMaterial({ color: 0xd45c8a, emissive: 0x5a1a2a, emissiveIntensity: 0.35, metalness: 0.1, roughness: 0.3 });

 

    const NPC = new THREE.Group();

    scene.add(NPC);

 

    // Membrane

    function createMembrane() {

      const g = new THREE.Group();

      const outerR = 14, innerR = 5.2;

      const ringGeo = new THREE.RingGeometry(innerR, outerR, 64);

      const outer = new THREE.Mesh(ringGeo, matMembrane);

      outer.rotation.x = -Math.PI/2; outer.position.y = 1.8;

      g.add(outer);

      const inner = outer.clone(); inner.position.y = -1.8; g.add(inner);

      g.add(new THREE.Mesh(new THREE.CylinderGeometry(innerR, innerR, 3.6, 48, 1, true), matMembrane));

      return g;

    }

    NPC.add(createMembrane());

 

    // 8-fold rings

    function createOctagonalRing(radius, height, thickness, y, material, name) {

      const group = new THREE.Group();

      group.name = name;

      for (let i = 0; i < 8; i++) {

        const angle = (i / 8) * Math.PI * 2;

        const x = Math.cos(angle) * radius;

        const z = Math.sin(angle) * radius;

        const box = new THREE.Mesh(new THREE.BoxGeometry(thickness*1.8, height, thickness), material);

        box.position.set(x, y, z);

        box.lookAt(0, y, 0);

        box.rotateY(Math.PI/2);

        group.add(box);

        const arc = new THREE.Mesh(

          new THREE.TorusGeometry(radius, thickness*0.45, 8, 12, Math.PI*2/8*0.85), material

        );

        arc.rotation.x = Math.PI/2;

        arc.rotation.z = angle + Math.PI/8;

        arc.position.y = y;

        group.add(arc);

      }

      return group;

    }

 

    const cytoRing = createOctagonalRing(6.2, 1.1, 0.85, 3.2, matScaffold, 'Cytoplasmic Ring (Nup88 / Nup214 / Nup358)');

    NPC.add(cytoRing);

    const nucRing  = createOctagonalRing(6.0, 1.1, 0.85, -3.2, matScaffold, 'Nuclear Ring');

    NPC.add(nucRing);

    const innerRing = createOctagonalRing(5.4, 1.6, 0.95, 0, matInner, 'Inner Ring (Nup93 / Nup155 / Nup205)');

    NPC.add(innerRing);

    const tmRing = createOctagonalRing(5.8, 0.6, 0.5, 0, new THREE.MeshStandardMaterial({

      color: 0x3a5a8a, metalness: 0.5, roughness: 0.4, transparent: true, opacity: 0.7

    }), 'Transmembrane NUPs (POM121 / NDC1 / GP210)');

    NPC.add(tmRing);

 

    // FG-Nups

    const fgGroup = new THREE.Group();

    fgGroup.name = 'FG-Nups (selective barrier)';

    for (let i = 0; i < 64; i++) {

      const angle = (i / 64) * Math.PI * 2 + Math.random()*0.2;

      const r = 1.2 + Math.random()*3.2;

      const points = [];

      for (let j = 0; j <= 12; j++) {

        const t = j/12;

        const y = -4 + t*8 + (Math.random()-0.5)*0.8;

        const wobble = Math.sin(t*Math.PI*3 + i)*0.35;

        points.push(new THREE.Vector3(Math.cos(angle)*(r+wobble), y, Math.sin(angle)*(r+wobble)));

      }

      const tube = new THREE.Mesh(

        new THREE.TubeGeometry(new THREE.CatmullRomCurve3(points), 16, 0.06+Math.random()*0.04, 4, false),

        matFG

      );

      fgGroup.add(tube);

    }

    NPC.add(fgGroup);

 

    // Filaments + Basket

    const filGroup = new THREE.Group();

    filGroup.name = 'Cytoplasmic Filaments';

    for (let i = 0; i < 8; i++) {

      const a = (i/8)*Math.PI*2;

      const base = new THREE.Vector3(Math.cos(a)*6.5, 3.8, Math.sin(a)*6.5);

      const tip  = base.clone().add(new THREE.Vector3(Math.cos(a)*2.5+(Math.random()-0.5), 4+Math.random()*2, Math.sin(a)*2.5+(Math.random()-0.5)));

      const curve = new THREE.CatmullRomCurve3([base, base.clone().lerp(tip,0.4).add(new THREE.Vector3(0,1.2,0)), tip]);

      filGroup.add(new THREE.Mesh(new THREE.TubeGeometry(curve,10,0.12,5,false), matFilament));

    }

    NPC.add(filGroup);

 

    const basketGroup = new THREE.Group();

    basketGroup.name = 'Nuclear Basket (Nup153 / TPR)';

    for (let i = 0; i < 8; i++) {

      const a = (i/8)*Math.PI*2;

      const base = new THREE.Vector3(Math.cos(a)*5.8, -3.8, Math.sin(a)*5.8);

      const mid  = new THREE.Vector3(Math.cos(a)*3.5, -6.5, Math.sin(a)*3.5);

      const tip  = new THREE.Vector3(Math.cos(a)*2.2, -8.2, Math.sin(a)*2.2);

      basketGroup.add(new THREE.Mesh(new THREE.TubeGeometry(new THREE.CatmullRomCurve3([base,mid,tip]),12,0.1,5,false), matBasket));

    }

    const distal = new THREE.Mesh(new THREE.TorusGeometry(2.4,0.15,8,32), matBasket);

    distal.rotation.x = Math.PI/2; distal.position.y = -8.3;

    basketGroup.add(distal);

    NPC.add(basketGroup);

 

    // Cargos

    const cargos = [];

    const cargoGroup = new THREE.Group();

    scene.add(cargoGroup);

 

    function spawnCargo(isImport) {

      const mesh = new THREE.Mesh(

        new THREE.SphereGeometry(0.28 + Math.random()*0.15, 10, 10),

        isImport ? matCargoIn.clone() : matCargoOut.clone()

      );

      const angle = Math.random()*Math.PI*2;

      const r = 1.5 + Math.random()*2.5;

      mesh.position.set(Math.cos(angle)*r, isImport ? 12+Math.random()*4 : -12-Math.random()*4, Math.sin(angle)*r);

      mesh.userData = { isImport, speed: 0.04+Math.random()*0.03, angle, active: true };

      cargoGroup.add(mesh);

      cargos.push(mesh);

    }

    for (let i=0;i<14;i++) spawnCargo(true);

    for (let i=0;i<10;i++) spawnCargo(false);

    // Click raycast

    const raycaster = new THREE.Raycaster();

    const mouse = new THREE.Vector2();

    window.addEventListener('click', (e) => {

      mouse.x = (e.clientX / window.innerWidth)*2 - 1;

      mouse.y = -(e.clientY / window.innerHeight)*2 + 1;

      raycaster.setFromCamera(mouse, camera);

      const hits = raycaster.intersectObjects(NPC.children, true);

      if (hits.length) {

        let obj = hits[0].object;

        while (obj && !obj.name) obj = obj.parent;

        if (obj?.name) document.getElementById('info-title').textContent = obj.name;

      }

    });

 

    // Animation

    function animate() {

      requestAnimationFrame(animate);

      time += 0.016 * animSpeed;

      controls.update();

      NPC.rotation.y = Math.sin(time * 0.15) * 0.07;

      fgGroup.children.forEach((t,i) => t.rotation.y = Math.sin(time*0.4 + i*0.3)*0.05);

 

      if (transportPlaying) {

        const info = stateInfo[currentState];

        cargos.forEach(c => {

          const ud = c.userData;

          const dir = ud.isImport ? -1 : 1;

          c.position.y += dir * ud.speed * animSpeed * info.cargoRate;

          if (Math.abs(c.position.y) < 6) {

            const targetR = 1.0 + Math.random()*1.5;

            const curR = Math.sqrt(c.position.x**2 + c.position.z**2);

            const newR = THREE.MathUtils.lerp(curR, targetR, 0.03);

            c.position.x = Math.cos(ud.angle)*newR;

            c.position.z = Math.sin(ud.angle)*newR;

          }

          if (ud.isImport && c.position.y < -14) {

            c.position.y = 12 + Math.random()*5;

            ud.angle = Math.random()*Math.PI*2;

            const r = 2+Math.random()*3;

            c.position.x = Math.cos(ud.angle)*r;

            c.position.z = Math.sin(ud.angle)*r;

          }

          if (!ud.isImport && c.position.y > 14) {

            c.position.y = -12 - Math.random()*5;

            ud.angle = Math.random()*Math.PI*2;

            const r = 2+Math.random()*3;

            c.position.x = Math.cos(ud.angle)*r;

            c.position.z = Math.sin(ud.angle)*r;

          }

        });

        if (currentState === 'cancer' && Math.random() < 0.012) spawnCargo(Math.random()>0.4);

      }

      renderer.render(scene, camera);

    }

 

    window.addEventListener('resize', () => {

      camera.aspect = window.innerWidth / window.innerHeight;

      camera.updateProjectionMatrix();

      renderer.setSize(window.innerWidth, window.innerHeight);

    });

 

    applyState('normal');

    animate();
